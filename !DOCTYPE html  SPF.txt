    function populateDistrictDropdowns() {
        const data = safeStorageGet('spf_districts_data', []);
        const jDist = document.getElementById('jDistrict');
        const vFinderDist = document.getElementById('userDistrictFinder');
        const admWaDist = document.getElementById('admWaDistrictSelect');

        let options = '<option value="">-- জিলা বাছনি কৰক --</option>';
        data.forEach(function(d) {
            options += '<option value="' + escapeHtml(d.name) + '">' + escapeHtml(d.name) + '</option>';
        });

        if (jDist) jDist.innerHTML = options;
        if (vFinderDist) vFinderDist.innerHTML = options;
        if (admWaDist) admWaDist.innerHTML = options;

        updateJoiningThanas();
        renderAdminDistList();
    }

    function updateJoiningThanas() {
        const dName = document.getElementById('jDistrict').value;
        const jThana = document.getElementById('jThana');
        const customThanaInput = document.getElementById('jThanaCustom');
        const data = safeStorageGet('spf_districts_data', []);
        const dist = data.find(function(d) { return d.name === dName; });

        if (customThanaInput) {
            customThanaInput.style.display = 'none';
            customThanaInput.value = '';
            customThanaInput.required = false;
        }

        if (dist && dist.thanas && dist.thanas.length > 0) {
            let opts = '<option value="">-- থানা বাছনি কৰক --</option>';
            dist.thanas.forEach(function(t) {
                opts += '<option value="' + escapeHtml(t) + '">' + escapeHtml(t) + '</option>';
            });
            opts += '<option value="OTHER">-- তালিকাৰ বাহিৰৰ থানা (অন্যান্য) --</option>';
            jThana.innerHTML = opts;
        } else if (dName) {
            jThana.innerHTML = '<option value="OTHER">-- আপোনাৰ থানাৰ নাম তলত লিখক --</option>';
            if (customThanaInput) {
                customThanaInput.style.display = 'block';
                customThanaInput.required = true;
            }
        } else {
            jThana.innerHTML = '<option value="">-- প্ৰথমে জিলা বাছক --</option>';
        }
    }

    function checkOtherThana(selectElem) {
        const customThanaInput = document.getElementById('jThanaCustom');
        if (selectElem.value === 'OTHER') {
            customThanaInput.style.display = 'block';
            customThanaInput.required = true;
            customThanaInput.focus();
        } else {
            customThanaInput.style.display = 'none';
            customThanaInput.required = false;
            customThanaInput.value = '';
        }
    }

            thanaSection.style.display = "block";

    function adminSaveVolunteer() {
        const editIdx = parseInt(document.getElementById('vEditIndex').value);
        const category = document.getElementById('vCategory').value;
        const name = document.getElementById('vName').value.trim();
        const role = document.getElementById('vRole').value.trim();
        const dist = document.getElementById('vDist').value.trim();
        const thana = document.getElementById('vThana').value.trim();
        const phone = document.getElementById('vPhone').value.trim();
        const wa = document.getElementById('vWa').value.trim();
        const fileInput = document.getElementById('vPhotoFile');
        const currentImg = document.getElementById('vPhotoCurrentBase').value;

        if (!name || !phone || !dist || !thana) {
            alert("অনুগ্ৰহ কৰি নাম, দায়িত্ব, জিলা, থানা আৰু ফোন নম্বৰ লিখক!");
            return;
        }

        function saveProcess(finalImg) {
            const list = safeStorageGet('spf_volunteers_data', []);
            const isLeadValue = (category === 'lead');

            if (editIdx >= 0 && editIdx < list.length) {
                list[editIdx].name = name;
                list[editIdx].role = role || (isLeadValue ? "জিলা মুৰব্বী" : "স্বেচ্ছাসেৱক");
                list[editIdx].dist = dist;
                list[editIdx].thana = thana;
                list[editIdx].phone = phone;
                list[editIdx].wa = wa || phone;
                list[editIdx].isLead = isLeadValue;
                if (finalImg) list[editIdx].img = finalImg;
                alert("তথ্য সফলতাৰে আপডেট কৰা হ'ল!");
            } else {
                list.push({
                    id: Date.now(),
                    name: name,
                    role: role || (isLeadValue ? "জিলা মুৰব্বী" : "স্বেচ্ছাসেৱক"),
                    dist: dist,
                    thana: thana,
                    phone: phone,
                    wa: wa || phone,
                    isLead: isLeadValue,
                    img: finalImg || OFFICIAL_LOGO
                });
                alert("সফলতাৰে যোগ কৰা হ'ল!");
            }

            safeStorageSet('spf_volunteers_data', list);
            adminCancelEditVolunteer();
            renderAdminVolunteersList();
        }

        if (fileInput.files && fileInput.files[0]) {
            compressImage(fileInput.files[0], 500, 0.7, function(compressedBase64) {
                saveProcess(compressedBase64);
            });
        } else {
            saveProcess(currentImg || OFFICIAL_LOGO);
        }
    }

    function toggleSidebar() {
        const drawer = document.getElementById('sidebarDrawer');
        const backdrop = document.getElementById('sidebarBackdrop');
        if (drawer.classList.contains('open')) {
            drawer.classList.remove('open');
            backdrop.style.display = 'none';
        } else {
            renderSidebarMenuItems();
            drawer.classList.add('open');
            backdrop.style.display = 'block';
        }
    }

    function renderSidebarMenuItems() {
        const list = document.getElementById('sidebarMenuList');
        const pages = safeStorageGet('spf_all_pages', []);

        let html = '<li class="sidebar-menu-item" onclick="navigateFromSidebar(\'home\')">' +
            '<i class="fa-solid fa-house"></i> <span>ঘৰ (Home)</span>' +
        '</li>' +
        '<li class="sidebar-menu-item" onclick="navigateFromSidebar(\'joining\')">' +
            '<i class="fa-solid fa-user-plus"></i> <span>সদস্য ভৰ্তি অভিযান</span>' +
        '</li>';

        pages.forEach(function(p) {

                '<i class="fa-solid fa-file-lines"></i> <span>' + escapeHtml(p.title) + '</span>' +
            '</li>';
        });
        list.innerHTML = html;
    }

    function navigateFromSidebar(target) {
        toggleSidebar();
        if (target === 'home') {
            window.scrollTo({ top: 0, behavior: 'smooth' });
        } else if (target === 'joining') {
            openJoiningModal();
        }
    }

    function openPageById(pageId) {
        const drawer = document.getElementById('sidebarDrawer');
        if (drawer.classList.contains('open')) toggleSidebar();

        const target = pages.find(function(p) { return p.id === pageId; });

        if (target) {
            document.getElementById('dynModalHeaderTitle').innerHTML = '<i class="fa-solid fa-file-lines"></i> ' + escapeHtml(target.title);
            document.getElementById('dynModalCover').src = target.cover || DEFAULT_COVER;
            document.getElementById('dynModalHeadline').innerText = target.headline;
            document.getElementById('dynModalContent').innerText = target.content;
            openModal('modalDynamicPage');
        }
    }

    // YouTube URL extraction helper
    function getYouTubeEmbedUrl(url) {
        if (!url) return '';
        let videoId = '';
        try {
            if (url.includes('youtu.be/')) {
                videoId = url.split('youtu.be/')[1].split(/[?#]/)[0];
            } else if (url.includes('youtube.com/watch')) {
                const urlParams = new URLSearchParams(new URL(url).search);
                videoId = urlParams.get('v');
            } else if (url.includes('youtube.com/embed/')) {
                videoId = url.split('youtube.com/embed/')[1].split(/[?#]/)[0];
            }
        } catch(e) {
            console.warn("YouTube parse error:", e);
        }
        return videoId ? 'https://www.youtube.com/embed/' + videoId : url;
    }

    function uploadAdminFile(input, storageKey) {
        if (input.files && input.files[0]) {
            const file = input.files[0];
            if (file.type.startsWith('image/')) {
                compressImage(file, 600, 0.7, function(base64) {
                    safeStorageSet(storageKey, base64);
                    alert("ফাইল সফলতাৰে সংৰক্ষণ হ'ল!");
                    loadMainApp();
                });
            } else {
                const reader = new FileReader();
                reader.onload = function(e) {
                    try {
                        safeStorageSet(storageKey, e.target.result);
                        alert("ফাইল সফলতাৰে সংৰক্ষণ হ'ল!");
                        loadMainApp();
                    } catch(err) {
                        alert("ফাইলটো অতি ডাঙৰ হোৱাৰ বাবে সংৰক্ষণ কৰিব পৰা নগ'ল।");
                    }
                };
                reader.readAsDataURL(file);
            }
        }
    }

                    '</div>' +
                '</div>' +
                '<div style="display:flex; gap:5px; margin-top:8px; border-top:1px dashed #cbd5e1; padding-top:6px;">' +
                    '<button onclick="setMemberStatus(' + idx + ', \'Approved\')" class="btn-mini" style="background:#15803d; color:white;"><i class="fa-solid fa-check"></i> অনুমোদন</button>' +
                    '<button onclick="setMemberStatus(' + idx + ', \'Cancelled\')" class="btn-mini" style="background:#d97706; color:white;"><i class="fa-solid fa-xmark"></i> বাতিল</button>' +

        });

    }

    function setMemberStatus(idx, newStatus) {
        const apps = safeStorageGet('spf_applications', []);
        apps[idx].status = newStatus;
        safeStorageSet('spf_applications', apps);
        loadAdminApplicants();
        let msg = (newStatus === 'Approved') ? "সদস্যজনক অনুমোদন জনোৱা হ'ল।" : (newStatus === 'Blocked' ? "সদস্যজনক ব্লক কৰা হ'ল!" : "সদস্যপদ বাতিল কৰা হ'ল।");
        alert(msg);
    }

    function adminDeleteMember(idx) {
        const apps = safeStorageGet('spf_applications', []);
        if (confirm("এই সদস্যৰ আবেদন সম্পূৰ্ণৰূপে ডিলেট কৰিবনে?")) {
            apps.splice(idx, 1);
            safeStorageSet('spf_applications', apps);
            loadAdminApplicants();
        }
    }

    function compressImage(file, maxWidth, quality, callback) {
        const reader = new FileReader();
        reader.onload = function(e) {
            const img = new Image();
            img.onload = function() {
                const canvas = document.createElement('canvas');
                let scale = Math.min(1, maxWidth / img.width);
                canvas.width = img.width * scale;
                canvas.height = img.height * scale;
                const ctx = canvas.getContext('2d');
                ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
                callback(canvas.toDataURL('image/jpeg', quality));
            };
            img.src = e.target.result;
        };
        reader.readAsDataURL(file);
    }

    function encodeFileToBase64(input, targetId) {
        if (input.files && input.files[0]) {
            const file = input.files[0];
            if (file.type.startsWith('image/')) {
                compressImage(file, 600, 0.7, function(base64) {
                    document.getElementById(targetId).value = base64;
                });
            } else {
                const reader = new FileReader();
                reader.onload = function(e) { document.getElementById(targetId).value = e.target.result; };
                reader.readAsDataURL(file);
            }
        }
    }
